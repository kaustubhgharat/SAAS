Step 2: Create Fields
Inside Product__c create these fields:
Field Label	Data Type	API Name
Product Name	Text	Product_Name__c
Serial No	Number	Serial_No__c
Manufacture Date	Date	Manufacture_Date__c
Expiry Date	Date	Expiry_Date__c


Step 3: Create Apex Class
Go to:
Setup → Apex Classes → New

public class ProductManagement {

    public Product__c productRec {get; set;}
    public Id editId {get; set;}

    // Constructor
    public ProductManagement() {

        productRec = new Product__c();
    }

    // Save or Update Product
    public PageReference saveProduct() {

        // ---------------- VALIDATIONS ----------------

        // Product Name Validation
        if(String.isBlank(productRec.Product_Name__c)) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Product Name is required'
                )
            );

            return null;
        }
		// Serial Number Validation
		if(productRec.Serial_No__c == null) {

    		ApexPages.addMessage(
        		new ApexPages.Message(
            		ApexPages.Severity.ERROR,
            		'Serial Number is required'
        		)
            );

    		return null;
		}

        // Manufacture Date Validation
        if(productRec.Manufacture_Date__c == null) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Manufacture Date is required'
                )
            );

            return null;
        }

        // Expiry Date Validation
        if(productRec.Expiry_Date__c == null) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Expiry Date is required'
                )
            );

            return null;
        }

        // Manufacture Date cannot be future date
        if(productRec.Manufacture_Date__c > Date.today()) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Manufacture Date cannot be future date'
                )
            );

            return null;
        }

        // Expiry Date must be greater than Manufacture Date
        if(productRec.Expiry_Date__c <=
           productRec.Manufacture_Date__c) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Expiry Date must be greater than Manufacture Date'
                )
            );

            return null;
        }

        // Duplicate Serial Number Validation
        List<Product__c> existingProduct = [

            SELECT Id
            FROM Product__c
            WHERE Serial_No__c = :productRec.Serial_No__c
            AND Id != :editId
        ];

        if(existingProduct.size() > 0) {

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.ERROR,
                    'Serial Number already exists'
                )
            );

            return null;
        }

        // ---------------- SAVE / UPDATE ----------------

        if(editId == null) {

            insert productRec;

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.CONFIRM,
                    'Product Added Successfully'
                )
            );

        } else {

            update productRec;

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.CONFIRM,
                    'Product Updated Successfully'
                )
            );

            editId = null;
        }

        productRec = new Product__c();

        return null;
    }

    // View Products
    public List<Product__c> getProducts() {

        return [

            SELECT Id,
                   Product_Name__c,
                   Serial_No__c,
                   Manufacture_Date__c,
                   Expiry_Date__c
            FROM Product__c
        ];
    }

    // Edit Product
    public PageReference editProduct() {

        Id pid = ApexPages.currentPage()
                 .getParameters()
                 .get('pid');

        productRec = [

            SELECT Id,
                   Product_Name__c,
                   Serial_No__c,
                   Manufacture_Date__c,
                   Expiry_Date__c
            FROM Product__c
            WHERE Id = :pid
        ];

        editId = pid;

        return null;
    }

    // Delete Product
    public PageReference deleteProduct() {

        Id pid = ApexPages.currentPage()
                 .getParameters()
                 .get('pid');

        Product__c p = [

            SELECT Id
            FROM Product__c
            WHERE Id = :pid
        ];

        delete p;

        ApexPages.addMessage(
            new ApexPages.Message(
                ApexPages.Severity.CONFIRM,
                'Product Deleted Successfully'
            )
        );

        return null;
    }
}

Step 4: Create Visualforce Page
Go to:
Setup → Visualforce Pages → New
Page Name:
ProductPage

<apex:page controller="ProductManagement">

    <apex:form>

        <apex:pageBlock title="Product Inventory Management">

            <apex:pageMessages/>

            <apex:pageBlockSection columns="1">

                <apex:inputText
                    value="{!productRec.Product_Name__c}"
                    label="Product Name"/>

                <apex:inputText
                    value="{!productRec.Serial_No__c}"
                    label="Serial No"/>

                <apex:inputField
                    value="{!productRec.Manufacture_Date__c}"/>

                <apex:inputField
                    value="{!productRec.Expiry_Date__c}"/>

            </apex:pageBlockSection>

            <apex:commandButton
                value="Save"
                action="{!saveProduct}"/>

        </apex:pageBlock>

        <br/>

        <apex:pageBlock title="Product List">

            <apex:pageBlockTable
                value="{!products}"
                var="p">

                <apex:column
                    value="{!p.Product_Name__c}"
                    headerValue="Product Name"/>

                <apex:column
                    value="{!p.Serial_No__c}"
                    headerValue="Serial No"/>

                <apex:column
                    value="{!p.Manufacture_Date__c}"
                    headerValue="Manufacture Date"/>

                <apex:column
                    value="{!p.Expiry_Date__c}"
                    headerValue="Expiry Date"/>

                <!-- Edit -->
                <apex:column headerValue="Edit">

                    <apex:commandLink
                        value="Edit"
                        action="{!editProduct}">

                        <apex:param
                            name="pid"
                            value="{!p.Id}"/>

                    </apex:commandLink>

                </apex:column>

                <!-- Delete -->
                <apex:column headerValue="Delete">

                    <apex:commandLink
                        value="Delete"
                        action="{!deleteProduct}">

                        <apex:param
                            name="pid"
                            value="{!p.Id}"/>

                    </apex:commandLink>

                </apex:column>

            </apex:pageBlockTable>

        </apex:pageBlock>

    </apex:form>

</apex:page>
