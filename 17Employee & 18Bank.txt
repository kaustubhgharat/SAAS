public class EmployeeManagement {

    // ADD Employee
    public static void addEmployee(String name, Integer empId,
                                   String email, Date birthDate,
                                   String dept) {

        Employee__c e = new Employee__c();

        e.Employee_Name__c = name;
        e.Employee_ID__c = empId;
        e.Email__c = email;
        e.Birth_Date__c = birthDate;
        e.Department__c = dept;

        insert e;

        System.debug('Employee Added Successfully');
    }

    // VIEW Employees
    public static void viewEmployees() {

        List<Employee__c> listEmp = [
            SELECT Employee_Name__c,
                   Employee_ID__c,
                   Email__c,
                   Birth_Date__c,
                   Department__c
            FROM Employee__c
        ];

        for(Employee__c e : listEmp) {

            System.debug(
                'Name: ' + e.Employee_Name__c +
                ' ID: ' + e.Employee_ID__c +
                ' Email: ' + e.Email__c +
                ' Birth Date: ' + e.Birth_Date__c +
                ' Department: ' + e.Department__c
            );
        }
    }

    // UPDATE Employee Email
    public static void updateEmployee(Integer empId, String newEmail) {

        Employee__c e = [
            SELECT Id, Email__c
            FROM Employee__c
            WHERE Employee_ID__c = :empId
            LIMIT 1
        ];

        e.Email__c = newEmail;

        update e;

        System.debug('Employee Updated Successfully');
    }

    // DELETE Employee
    public static void deleteEmployee(Integer empId) {

        Employee__c e = [
            SELECT Id
            FROM Employee__c
            WHERE Employee_ID__c = :empId
            LIMIT 1
        ];

        delete e;

        System.debug('Employee Deleted Successfully');
    }
}


// Add Employee

    'IT'
);


// View Employees
EmployeeManagement.viewEmployees();

    101,
