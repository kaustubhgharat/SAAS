public with sharing class EmployeeController {

    @AuraEnabled
    public static String saveEmployee(
        String empName,
        Integer empId,
        Decimal salary,
        String email,
        String department,
        Date joiningDate
    ) {

        // Name Validation
        if(String.isBlank(empName) || empName.length() < 3) {

            return 'Employee Name must contain at least 3 characters';
        }

        // Employee ID Validation
        if(empId <= 0) {

            return 'Employee ID must be greater than 0';
        }

        // Unique Employee ID
        List<Emp__c> empList = [

            SELECT Id
            FROM Emp__c
            WHERE Emp_ID__c = :empId
        ];

        if(empList.size() > 0) {

            return 'Employee ID already exists';
        }

        // Salary Validation
        if(salary < 10000 || salary > 500000) {

            return 'Salary must be between 10,000 and 500,000';
        }

        // Email Validation
        Pattern pattern = Pattern.compile(
            '^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$'
        );

        Matcher matcher = pattern.matcher(email);

        if(!matcher.matches()) {

            return 'Invalid Email Format';
        }

        // Department Validation
        if(String.isBlank(department)) {

            return 'Please select department';
        }

        // Joining Date Validation
        if(joiningDate > Date.today()) {

            return 'Joining Date cannot be future date';
        }

        // Save Record
        Emp__c emp = new Emp__c();

        emp.Emp_Name__c = empName;
        emp.Emp_ID__c = empId;
        emp.Salary__c = salary;
        emp.Email__c = email;
        emp.Department__c = department;
        emp.Joining_Date__c = joiningDate;

        insert emp;

        return 'Employee Saved Successfully';
    }
}

import { LightningElement, wire, track } from 'lwc';

import saveEmployeeData
from '@salesforce/apex/EmployeeController.saveEmployee';

import getEmployees
from '@salesforce/apex/EmployeeController.getEmployees';

import { ShowToastEvent }
from 'lightning/platformShowToastEvent';

import { refreshApex }
from '@salesforce/apex';

export default class EmployeeManagement extends LightningElement {

    empName;
    empId;
    salary;
    email;
    department;
    joiningDate;

    @track employeeList;

    wiredResult;

    columns = [

        { label: 'Name', fieldName: 'Emp_Name__c' },
        { label: 'Employee ID', fieldName: 'Emp_ID__c' },
        { label: 'Salary', fieldName: 'Salary__c' },
        { label: 'Email', fieldName: 'Email__c' },
        { label: 'Department', fieldName: 'Department__c' },
        { label: 'Joining Date', fieldName: 'Joining_Date__c' }

    ];

    deptOptions = [

        { label: 'IT', value: 'IT' },
        { label: 'HR', value: 'HR' },
        { label: 'Sales', value: 'Sales' }

    ];

    @wire(getEmployees)
    wiredEmployees(result) {

        this.wiredResult = result;

        if(result.data) {

            this.employeeList = result.data;
        }
    }

    handleName(event) {

        this.empName = event.target.value;
    }

    handleId(event) {

        this.empId = event.target.value;
    }

    handleSalary(event) {

        this.salary = event.target.value;
    }

    handleEmail(event) {

        this.email = event.target.value;
    }

    handleDepartment(event) {

        this.department = event.target.value;
    }

    handleDate(event) {

        this.joiningDate = event.target.value;
    }

    saveEmployee() {

        saveEmployeeData({

            empName: this.empName,
            empId: this.empId,
            salary: this.salary,
            email: this.email,
            department: this.department,
            joiningDate: this.joiningDate

        })

        .then(result => {

            this.dispatchEvent(

                new ShowToastEvent({

                    title: 'Success',
                    message: result,
                    variant: 'success'

                })
            );

            refreshApex(this.wiredResult);
        })

        .catch(error => {

            this.dispatchEvent(

                new ShowToastEvent({

                    title: 'Error',
                    message: error.body.message,
                    variant: 'error'

                })
            );
        });
    }
}
