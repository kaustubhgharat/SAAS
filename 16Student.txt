   

 

public class StudentManagement {

    public Student__c studentRec {get; set;}
    public Id editId {get; set;}

    public StudentManagement() {

        studentRec = new Student__c();
    }

    // Save or Update Student
    public PageReference saveStudent() {

        if(editId == null) {

            insert studentRec;

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.CONFIRM,
                    'Student Added Successfully'
                )
            );

        } else {

            update studentRec;

            ApexPages.addMessage(
                new ApexPages.Message(
                    ApexPages.Severity.CONFIRM,
                    'Student Updated Successfully'
                )
            );

            editId = null;
        }

        studentRec = new Student__c();

        return null;
    }

    // View Students
    public List<Student__c> getStudents() {

        return [
            SELECT Id,
                   Student_Name__c,
                   Roll_Number__c,
                   Class__c,
                   Mobile_Number__c
            FROM Student__c
        ];
    }

    // Edit Student
    public PageReference editStudent() {

        Id sid = ApexPages.currentPage()
                 .getParameters()
                 .get('sid');

        studentRec = [
            SELECT Id,
                   Student_Name__c,
                   Roll_Number__c,
                   Class__c,
                   Mobile_Number__c
            FROM Student__c
            WHERE Id = :sid
        ];

        editId = sid;

        return null;
    }

    // Delete Student
    public PageReference deleteStudent() {

        Id sid = ApexPages.currentPage()
                 .getParameters()
                 .get('sid');

        Student__c s = [
            SELECT Id
            FROM Student__c
            WHERE Id = :sid
        ];

        delete s;

        return null;
    }
}
